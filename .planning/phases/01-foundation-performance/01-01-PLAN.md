**Fingerprint function (replaces lines 61-68):**
```zsh
# D-01: usa tool --version em vez de command -v (path-based)
# Captura a string de versão; se o tool não existe, usa "missing"
# D-03: ferramentas ausentes não bloqueiam — são marcadas como "missing"
_zsh_gen_fingerprint() {
  local fp="" tool version
  for tool in "${_TOOLS_WATCHED[@]}"; do
    version=$(command "$tool" --version 2>/dev/null | head -1 || echo "missing")
    fp+="${tool}=${version};"
  done
  # cksum para hash determinístico; fallback para string bruta se cksum falhar
  print -n "$fp" | cksum | cut -d' ' -f1 2>/dev/null || print -n "$fp"
}
```

**Cache rebuild function (replaces lines 70-91):**
```zsh
# Rebuild atômico: escreve em tmp, mv no final (nunca escreve direto no cache)
# D-07: arquivo monolítico — todas as ferramentas em um único source
# D-08: fingerprint na linha 1 para validação atômica
_zsh_build_plugin_cache() {
  local tmp=$(mktemp) || return 1
  trap "rm -f '$tmp'" INT TERM  # preserva o padrão atômico mktemp→mv
  printf '# zsh_plugin_cache fingerprint: %s\n' "$(_zsh_gen_fingerprint)" > "$tmp"

  # zoxide init (cacheia a saída completa do init para evitar subprocess)
  if command -v zoxide &>/dev/null; then
    printf '\n# --- zoxide ---\n' >> "$tmp"
    zoxide init zsh >> "$tmp" 2>/dev/null
  fi

  # fzf key bindings + completion (source paths do Fedora)
  if command -v fzf &>/dev/null; then
    printf '\n# --- fzf ---\n' >> "$tmp"
    [[ -f /usr/share/fzf/shell/key-bindings.zsh ]] && \
      printf 'source /usr/share/fzf/shell/key-bindings.zsh\n' >> "$tmp"
    [[ -f /usr/share/zsh/site-functions/_fzf ]] && \
      printf 'source /usr/share/zsh/site-functions/_fzf\n' >> "$tmp"
  fi

  # eza (D-02: monitorado no fingerprint, mas sem init — alias-only tool)
  if command -v eza &>/dev/null; then
    printf '\n# --- eza: sem init necessário (alias-only) ---\n' >> "$tmp"
  fi

  mv "$tmp" "$_PLUGIN_CACHE"  # mv atômico — preserva o padrão existente
}
```

**Cache validation + flock locking (replaces lines 93-101):**
```zsh
# Validação do cache com flock (D-04: previne rebuilds concorrentes)
# D-09: rebuild síncrono quando fingerprint não bate
# D-10: cache corrompido/ausente → fallback silencioso
# D-12: rebuild silencioso — sem mensagens no terminal

if [[ -f "$_PLUGIN_CACHE" ]]; then
  _zsh_current_fp=$(_zsh_gen_fingerprint)
  # D-08: fingerprint está na linha 1 do cache
  _zsh_cached_fp=$(sed -n '1s/# zsh_plugin_cache fingerprint: //p' "$_PLUGIN_CACHE" 2>/dev/null)

  if [[ "$_zsh_current_fp" != "$_zsh_cached_fp" ]]; then
    # Fingerprint divergente — cache está stale. Tenta rebuild com lock.
    local _lockfile="${_PLUGIN_CACHE}.lock"
    # D-04: flock -n é non-blocking — se outro shell já está rebuildando,
    # usa o cache existente (pode estar alguns segundos desatualizado)
    touch "$_lockfile" 2>/dev/null
    zmodload zsh/system 2>/dev/null  # zsystem flock (fallback se util-linux flock ausente)
    if command -v flock &>/dev/null; then
      # flock do util-linux — kernel-enforced, libera ao fechar fd
      exec {_cachelock_fd}<>"$_lockfile" 2>/dev/null
      if [[ -n ${_cachelock_fd:-} ]] && flock -n "$_cachelock_fd" 2>/dev/null; then
        _zsh_build_plugin_cache
        exec {_cachelock_fd}>&-  # fecha fd → libera lock
      fi
    elif zmodload -e zsh/system && zsystem flock -t 0 "$_lockfile" 2>/dev/null; then
      # zsystem flock (builtin do Zsh) como fallback
      _zsh_build_plugin_cache
      zsystem flock -u "$_lockfile" 2>/dev/null
    else
      # Sem mecanismo de lock — rebuild sem proteção (melhor que travar)
      _zsh_build_plugin_cache
    fi
  fi
else
  # Cache não existe — primeiro boot ou foi removido
  _zsh_build_plugin_cache
fi
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Cache file → Shell | `zsh_plugins_init.zsh` is sourced directly into the shell process. Its contents (zoxide init output, fzf key bindings) execute with full user privileges. |
| Tool binaries → Cache | `zoxide init zsh` and fzf key-binding source commands produce content that enters the cache file. The tool binaries must be trusted. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-01 | Tampering | Cache file (zsh_plugins_init.zsh) | mitigate | Atomic mktemp→mv pattern: cache is never written in-place. Partial writes produce no cache file at all (worst case: rebuild on next start). |
| T-01-02 | Tampering | Cache file via tool output | accept | `zoxide init zsh` output is trusted — the tool binary is installed via DNF (signed packages). If the binary is compromised, the attacker already has code execution. |
| T-01-03 | Denial of Service | Lock file (zsh_plugins_init.zsh.lock) | mitigate | `flock -n` is non-blocking — if lock is held, fall through to use existing cache. A stale lock file does not block startup. |
| T-01-04 | Information Disclosure | XDG env vars | accept | XDG vars point to standard user home paths. No sensitive data is exposed. Fallback to defaults if vars are unset. |
</threat_model>
