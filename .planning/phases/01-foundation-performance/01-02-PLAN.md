```zsh
# ==============================================================================
# FILTRO DE SEGURANÇA DO HISTÓRICO — Bloqueia credenciais e material sensível
# D-20: sem logging de comandos bloqueados (descarte silencioso)
# D-26: captura URL tokens (https://token@host), --token <value>, --password <value>, -p <value>
# D-27: captura SSH key material (ssh -i, ssh-add, ssh-keygen) e GPG passphrase operations
# Pitfall 6: early-exit para comandos > 4096 chars (pasted scripts, heredocs)
# ==============================================================================
zshaddhistory() {
  # Pitfall 6: comandos muito longos (paste de scripts, heredocs) pulam o filtro
  # para evitar overhead de regex em strings de 10K+ caracteres
  [[ ${#1} -gt 4096 ]] && return 0

  local upper="${1:u}"

  # --- Bloqueio de padrões de atribuição (comportamento existente, mantido) ---
  # Ex: export TOKEN=abc123, PASSWORD=secret, API_KEY=val
  [[ "$upper" =~ (TOKEN|SECRET|PASSWORD|PASSWD|API_KEY|PRIVATE_KEY|CREDENTIAL|ACCESS_KEY)[[:space:]]*= ]] && return 1

  # --- D-26: Bloqueio de credenciais embutidas em URL ---
  # Ex: git clone https://token@github.com/user/repo.git
  # Ex: curl https://user:password@api.example.com/endpoint
  [[ "$upper" =~ HTTPS?://[^[:space:]]*@[^[:space:]]+ ]] && return 1

  # --- D-26: Bloqueio de credenciais via flags longas ---
  # Ex: some-tool --token ghp_xxxxxxxxxxxx
  # Ex: some-tool --password mysecret
  [[ "$upper" =~ --TOKEN[[:space:]]+[^[:space:]]+ ]] && return 1
  [[ "$upper" =~ --PASSWORD[[:space:]]+[^[:space:]]+ ]] && return 1

  # --- D-26: Bloqueio de credenciais via flag curta -p ---
  # Ex: mysql -p mypassword (cuidado: não captura -p sem valor)
  [[ "$upper" =~ (^|[[:space:]])-P[[:space:]]+[^[:space:]-][^[:space:]]* ]] && return 1

  # --- D-27: Bloqueio de material de chave SSH ---
  # Ex: ssh -i ~/.ssh/id_rsa user@host
  # Ex: ssh-add ~/.ssh/id_rsa
  # Ex: ssh-keygen -t rsa -f ~/.ssh/new_key
  [[ "$upper" =~ (SSH[[:space:]]+-I|SSH-ADD[[:space:]]|SSH-KEYGEN) ]] && return 1

  # --- D-27: Bloqueio de operações com passphrase GPG ---
  # Ex: gpg --passphrase secret --decrypt file.gpg
  # Ex: gpg --batch --passphrase-fd 0
  [[ "$upper" =~ GPG[[:space:]].*--PASS ]] && return 1

  return 0
}
```

CRITICAL RULES:
- Keep the function name as `zshaddhistory` (Zsh hook convention — must match exactly)
- Keep the comment header documenting D-20, D-26, D-27, and Pitfall 6
- The `return 0` at the end must be preserved (commands that pass all checks ARE saved to history)
- Do NOT add any echo/printf output — this is a hook, output would break shell startup (D-20: silent discard)
- The early-exit length check (`[[ ${#1} -gt 4096 ]] && return 0`) must be FIRST inside the function body per Pitfall 6

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| User input → History filter | Every command the user types passes through `zshaddhistory()` before being saved to HISTFILE. Maliciously crafted command strings could attempt regex bypass. |
| HISTFILE → Shell sessions | `~/.zsh_history` is readable by the user. Other processes with user privileges could read it. The filter prevents sensitive data from reaching this file. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-05 | Information Disclosure | `zshaddhistory()` hook — assignment patterns | mitigate | Case-insensitive regex blocks `TOKEN=`, `SECRET=`, `PASSWORD=`, `PASSWD=`, `API_KEY=`, `PRIVATE_KEY=`, `CREDENTIAL=`, `ACCESS_KEY=` followed by `=`. |
| T-01-06 | Information Disclosure | `zshaddhistory()` hook — URL-embedded credentials | mitigate | Regex `HTTPS?://[^[:space:]]*@[^[:space:]]+` blocks `https://user:pass@host` and `https://token@host` patterns (D-26). |
| T-01-07 | Information Disclosure | `zshaddhistory()` hook — flag-based credentials | mitigate | Regex blocks `--token <value>`, `--password <value>`, and `-p <value>` patterns (D-26). |
| T-01-08 | Information Disclosure | `zshaddhistory()` hook — SSH key material | mitigate | Regex blocks `ssh -i`, `ssh-add`, and `ssh-keygen` commands that may expose key paths (D-27). |
| T-01-09 | Information Disclosure | `zshaddhistory()` hook — GPG passphrase operations | mitigate | Regex blocks `gpg ... --passphrase ...` and `gpg ... --passphrase-fd` commands (D-27). |
| T-01-10 | Denial of Service | `zshaddhistory()` hook — regex performance | mitigate | Early-exit on commands > 4,096 characters (Pitfall 6). Pasted scripts and heredocs bypass the filter entirely rather than consuming CPU on regex matching. |
| T-01-11 | Elevation of Privilege | `zsh/.zshrc` — theme loading order | accept | P10k theme source runs with user privileges. Ordering violation (theme loaded too early) only affects visual prompt appearance, not security. No privilege boundary crossed. |

## Accepted Risks

| Threat ID | Rationale |
|-----------|-----------|
| T-01-11 | Theme loading order is a correctness concern (prompt appearance), not a security boundary. Incorrect ordering cannot be exploited for privilege escalation or information disclosure. |
</threat_model>
